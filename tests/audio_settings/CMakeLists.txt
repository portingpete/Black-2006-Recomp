cmake_minimum_required(VERSION 3.20)
project(black_audio_settings C)
add_executable(audio-settings-test test_audio_settings.c)
target_include_directories(audio-settings-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
target_compile_definitions(audio-settings-test PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(audio-settings-test PRIVATE /W4 /WX)
else()
    target_compile_options(audio-settings-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME audio-settings COMMAND audio-settings-test)
